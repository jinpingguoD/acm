也就是树上面任意两点之间的距离求相对应的最大值
那么相对应的也可以求出来这棵树上面距离最远的两个点
也可以求出这棵书上面任意一个点在这棵树上面最远点的距离（其实就是直径那两个点选择一个距离最大的）
```cpp
int BFS(int n) {一遍BFS，目的是找出来当前这个节点对应的最远的点
	vector<vector<pair<int, int>>>l1;
	int l2[100000];
	queue<int>l3;
	memset(l2, -1, sizeof(l2));
	l2[n] = 0;
	l3.push(n);
	int k1 = n;
	while (!l3.empty()) {
		int k2 = l3.front();
		l3.pop();
		for (auto it : l1[k2]) {
			if (l2[it.first] == -1) {
				l2[it.first] = l2[k2] + it.second;
				if (l2[it.first] > l2[k1]) {
					记录最远点
					k1 = it.first;
				}
				l3.push(it.first);
			}
		}
	}
	return k1;
}
int main(){
	int k1=BFS(1);
	第一遍随便找一个点去进行BFS
	这样就找出来直径上面两个点其中之一
	int k2=BFS(k1);
	然后以这个点再次进行一边BFS，这样就找到了另一个点
}
```
