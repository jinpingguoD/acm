首先来看看什么是快速幂，如果现在有问题要求求出相对应的a的b次方，那么直接用pow函数显得太慢，这时候还是用二进制分解的思想，可以把这一过程转换为二进制拆解，也就是快速幂
代码：
```cpp
ll Pow(ll a, ll b) {
    ll r = 1;
    也就是把b看成二进制，这样就会发现实际上b也就是不同个1在不同位置上组成的
    那么由这个思想延伸过来，把b逐渐往右边进，与此同时a不断乘以自身
    这样当b最右边一位已经是1的时候就直接乘a
    也就是把b成功拆解
    while(b != 0) {
        if (b % 2 == 1) {
            r *= a;
        }
        a *= a; b /= 2;
    }
    return r;
}
```
由上面的概念延伸过来，因为不同矩阵之间相乘遵守乘法结合律，那么a矩阵的b次幂也可以换成上面的形式，也就是把乘法的形式改变一下就行了
代码：
```cpp
struct Matrix {
	//关于矩阵这个结构的定义
    int n;//矩阵的边
    vector<vector<ll>> a;
    Matrix(int n) : n(n), a(n + 2, vector<ll>(n + 2, 0)) {}//矩阵初始化
    static Matrix identity(int n) {//一个单位矩阵的快速构建
        Matrix I(n);
        for (int i = 1; i <= n; ++i) I.a[i][i] = 1;
        return I;
    }
};
Matrix F1(const Matrix& A, const Matrix& B) {//矩阵乘法
    int n = A.n;
    Matrix C(n);
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (A.a[i][j] == 0) continue;
            for (int k = 1; k <= n; ++k) {
                C.a[i][k] = (C.a[i][k] + A.a[i][j] * B.a[j][k]) % MOD;
            }
        }
    }
    return C;
}
Matrix F2(Matrix base, ll b) {//矩阵快速幂本身
    int n = base.n;
    Matrix k1 = Matrix::identity(n);
    while (b != 0) {
        if (b % 2 == 1) {
            k1 = F1(base, k1);
        }
        base = F1(base, base), b /= 2;
    }
    return k1;
}
ll Pow(ll a, ll b) {
    ll r = 1;
    while(b != 0) {
        if (b % 2 == 1) {
            r *= a;
        }
        a *= a; b /= 2;
    }
    return r;
}
```
