一个能够快速实现查询数组前缀和的数据结构，当然功能也包括实现边修改边查询这个功能，也就是支持动态更新与查询数组区间相对应的值
建树的时间复杂度O（n），空间复杂度O（n）
原理就是把原数组根据lowbit看成一个个确定的区间，也就是不同的下标对应着不同的区间包括着各种元素，比如下标1对应的就是[1,1]这个区间的元素，下标4对应的就是[1，4]这个区间的元素
先给出来一个只支持单点修改的树状数组的代码
```cpp
vector<int>l1;//原数组
vector<int>l2;//树状数组
int n;//数组长度
int lowbit(int n) {
    return n & (-n);
}
int Find(int n) {//实现查询
    ll sum = 0;
    while (n > 0) {
        sum += l2[n];
        n -= lowbit(n);
    }
    //如果实现某一个区间[l,r]只需要把相对应的前缀和(l-1,r)求出来相减就行了
    return sum;
}
void add(int a,int b) {//单点加和 也就是下标a加上b
    while (a <= n) {
        l2[a] += b;
        a += lowbit(a);
    }
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0); cout.tie(0);
    l2 = l1;//开始的时候先把原数组复制一遍
    rep(i, 1, n) {//O(n)建树
        int k1 = i + lowbit(i);
        if (k1 <= n) {
            l2[k1] += l2[i];
        }
    }
}
```
要想实现区间修改（也就是在对原数组进行区间上面的修改），这时候需要两个树状数组。
首先可知对于差分数组来说，区间修改只需要改变左右两端的端点的值就行了，因此速度较快，也以此可以看为是一个单点修改的操作，故可以用树状数组先来维护原数组的差分数组。
那么这个时候因为原数组的每一个值对应的都是差分数组的前缀和，而求原数组的前缀和也相当于对差分数组求和的基础之上在进行求和，这一部分可由数学证明，最终的值等同于一个树状数组记录di* i的前缀和（d表示差分数组）
因此区间修改以及查询需要用到两个树状数组进行操作
下面给出相对应的代码
```cpp
int n;
vector<int> a;// 原数组
vector<int> d;// 差分数组
vector<int> l1;// 树状数组1：维护 d[i]
vector<int> l2;// 树状数组2：维护 d[i] * i
int lowbit(int x) {
    return x & (-x);
}
// 单点修改：在树状数组 tree 的下标 idx 处增加 val
void add(vector<int>& tree, int idx, int val) {
    while (idx <= n) {
        tree[idx] += val;
        idx += lowbit(idx);
    }
}
// 区间加：将原数组 [l, r] 区间每个数增加 val
void range_add(int l, int r, int val) {
    // 修改差分数组 d[l] += val
    add(l1, l, val);
    add(l2, l, val * l);
    // 修改差分数组 d[r+1] -= val（若 r+1 <= n）
    if (r + 1 <= n) {
        add(l1, r + 1, -val);
        add(l2, r + 1, -val * (r + 1));
    }
}

// 前缀查询：求树状数组 tree 的前缀和（1~idx）
int sum(const vector<int>& tree, int idx) {
    int res = 0;
    while (idx > 0) {
        res += tree[idx];
        idx -= lowbit(idx);
    }
    return res;
}
// 查询原数组前 x 个元素的总和 S(x)
int prefix_sum(int x) {
    // S(x) = (x+1) * sum(l1, x) - sum(l2, x)
    return (x + 1) * sum(l1, x) - sum(l2, x);
}
// 区间查询：查询原数组 [l, r] 的总和
int range_query(int l, int r) {
    return prefix_sum(r) - prefix_sum(l - 1);
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0); cout.tie(0);
    // 构建差分数组 d
    d[1] = a[1];
    rep(i, 2, n) d[i] = a[i] - a[i - 1];
    // 初始化树状数组的原始数据
    rep(i, 1, n) {
        l1[i] = d[i];
        l2[i] = d[i] * i;
    }
    // O(n) 线性建树
    rep(i, 1, n) {
        int k1 = i + lowbit(i);
        if (k1 <= n) {
            l1[k1] += l1[i];
            l2[k1] += l2[i];
        }
    }
    return 0;
}
```
