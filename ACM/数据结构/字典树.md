字典树是关于string的一个快速的存储结构，本身的原理就是对于一个字符串，把他的每一个都添加到一个树中，比如一个“abc”与一个“ab”，实现先添加一个a b然后添加一个c，也就是有相同前缀的字符串放到一起，防止重复添加，然后后面的多余的再次添加。
字典树可以实现查询字符串是否存在，以及查询前缀是否存在，当然在这里面也有专门的函数来判断是前缀存在还是字符串存在
字典树原本应该是指针写出来的（毕竟是树），在这里用vector模拟，如果当下的这一个点是存在的那么将字符串接着循环下去。如果不存在，那么新添加一个vector元素，并把这个vector元素的下标给到当前的这个元素，这样做的好处就是防止内存泄露。
对于其中的每一个vector元素都用接近于静态数组的array存储，看具体的字符串的内容来决定此数组的长度（注意，如果题目没有要求那么多但还是把静态数组开的很大，很容易MLE），然后一个getId函数专门用来处理相对应的字符转换为数字。
至于为什么不能把array换成int，以至于每一个节点都只用一个int类型就能存储，这是因为假如对于一个’a‘节点，我想要知道下一步是否由b c节点，这时候如果只有一个int类型是不能实现的，同样的可能指针的写法能实现，但那样遍历就非常麻烦了。

```cpp
struct Trie {
    vector<array<int, 62>> nxt;
    vector<int> cnt;
    vector<bool> isEnd;
    Trie() {//初始化
        nxt.push_back({});
        cnt.push_back(0);
        isEnd.push_back(false);
    }
    int getId(char c) const {//对不同字符进行判断
        if (isdigit(c)) return c - '0';
        if (islower(c))return c - 'a' + 10;
        if (isupper(c))return c - 'A' + 26;
    }
    void insert(const string& s) {//插入单词
        int u = 0;
        for (char c : s) {
            int v = getId(c);
            if (nxt[u][v] == 0) {
	            //如果当前没有找到这个单词
	            //那么就新添加一个元素,然后把下标指向这个元素,表示这个元素存在
                nxt[u][v] = (int)nxt.size();
                nxt.push_back({});
                cnt.push_back(0);
                isEnd.push_back(false);
            }
            u = nxt[u][v];
            cnt[u]++;
        }
        isEnd[u] = true;//遇见字符串末尾就自动判断对,表示这是字符串末尾
    }
    bool find(const string& s) const {//查找单词是否存在
        int u = 0;
        for (char c : s) {
            int v = getId(c);
            if (nxt[u][v] == 0) return false;
            u = nxt[u][v];
        }
        return isEnd[u];
    }
    bool startsWith(const string& s) const {//是否有关于s前缀的单词
        int u = 0;
        for (char c : s) {
            int v = getId(c);
            if (nxt[u][v] == 0) return false;
            u = nxt[u][v];
        }
        return true;
    }
    int prefixCount(const string& s) const {//统计以s为前缀的单词数量
        int u = 0;
        for (char c : s) {
            int v = getId(c);
            if (nxt[u][v] == 0) return 0;
            u = nxt[u][v];
        }
        return cnt[u];
    }
    void erase(const string& s) {
	    //删除,不过是数字意义上的删除
	    //如果实际上的清空内存很耗时,而且可能倒是
        if (!find(s)) return;
        int u = 0;
        for (char c : s) {
            int v = getId(c);
            u = nxt[u][v];
            cnt[u]--;
        }
        isEnd[u] = false;
    }
    void clear() {//清空
        nxt.clear();
        cnt.clear();
        isEnd.clear();
        nxt.push_back({});
        cnt.push_back(0);
        isEnd.push_back(false);
    }
};
```
